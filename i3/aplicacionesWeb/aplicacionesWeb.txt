Crea:

Si la carpeta todavía no existe:

Pega todo esto:

#!/bin/bash

# ============================================================
# RT-Linux-Theme — WebApps Installer
# Aplicaciones web para Brave + Rofi + i3
#
# Iconos: Simple Icons
# Navegador: Brave
# Lanzador: Rofi
# ============================================================

set -e

# ------------------------------------------------------------
# CONFIGURACIÓN
# ------------------------------------------------------------

THEME_DIR="$HOME/RT-Linux-Theme"
WEBAPPS_DIR="$THEME_DIR/webapps"

ICON_DIR="$WEBAPPS_DIR/icons"
DESKTOP_DIR="$WEBAPPS_DIR/applications"

SYSTEM_APPS="$HOME/.local/share/applications"

BASE_URL="https://cdn.simpleicons.org"

# ------------------------------------------------------------
# COLORES DE TERMINAL
# ------------------------------------------------------------

RESET="\033[0m"
BOLD="\033[1m"
GREEN="\033[32m"
TERRACOTTA="\033[31m"
YELLOW="\033[33m"
CYAN="\033[36m"

# ------------------------------------------------------------
# FUNCIONES
# ------------------------------------------------------------

info() {
    echo -e "${CYAN}[INFO]${RESET} $1"
}

ok() {
    echo -e "${GREEN}[ OK ]${RESET} $1"
}

warn() {
    echo -e "${YELLOW}[WARN]${RESET} $1"
}

error() {
    echo -e "${TERRACOTTA}[ERROR]${RESET} $1"
}

# ------------------------------------------------------------
# CABECERA
# ------------------------------------------------------------

clear

echo
echo -e "${BOLD}==============================================${RESET}"
echo -e "${BOLD} RT-Linux-Theme — WebApps${RESET}"
echo -e "${BOLD}==============================================${RESET}"
echo
echo " Brave + Rofi + i3"
echo " Iconos SVG + aplicaciones .desktop"
echo

# ------------------------------------------------------------
# COMPROBAR BRAVE
# ------------------------------------------------------------

if command -v brave-browser >/dev/null 2>&1; then

    BRAVE="brave-browser"

elif command -v brave >/dev/null 2>&1; then

    BRAVE="brave"

else

    error "No se encontró Brave."

    echo
    echo "Instala Brave y vuelve a ejecutar este script."
    echo

    exit 1

fi

ok "Brave encontrado: $BRAVE"

# ------------------------------------------------------------
# COMPROBAR CURL
# ------------------------------------------------------------

if ! command -v curl >/dev/null 2>&1; then

    error "curl no está instalado."

    echo
    echo "Instálalo con:"
    echo
    echo "sudo apt install curl"
    echo

    exit 1

fi

ok "curl encontrado"

# ------------------------------------------------------------
# CREAR ESTRUCTURA
# ------------------------------------------------------------

info "Creando estructura de carpetas..."

mkdir -p "$ICON_DIR"
mkdir -p "$DESKTOP_DIR"
mkdir -p "$SYSTEM_APPS"

ok "Estructura creada"

echo
echo "WebApps:"
echo "  $WEBAPPS_DIR"
echo
echo "Iconos:"
echo "  $ICON_DIR"
echo
echo "Archivos .desktop:"
echo "  $DESKTOP_DIR"
echo
echo "Instalación Rofi:"
echo "  $SYSTEM_APPS"
echo

# ------------------------------------------------------------
# DEFINICIÓN DE WEBAPPS
# ------------------------------------------------------------

declare -A URLS

URLS[youtube]="https://www.youtube.com"
URLS[gmail]="https://mail.google.com"
URLS[whatsapp]="https://web.whatsapp.com"
URLS[spotify]="https://open.spotify.com"
URLS[instagram]="https://www.instagram.com"
URLS[telegram]="https://web.telegram.org"
URLS[discord]="https://discord.com/app"
URLS[github]="https://github.com"

declare -A NAMES

NAMES[youtube]="YouTube"
NAMES[gmail]="Gmail"
NAMES[whatsapp]="WhatsApp Web"
NAMES[spotify]="Spotify"
NAMES[instagram]="Instagram"
NAMES[telegram]="Telegram Web"
NAMES[discord]="Discord"
NAMES[github]="GitHub"

declare -A COMMENTS

COMMENTS[youtube]="YouTube en Brave"
COMMENTS[gmail]="Gmail en Brave"
COMMENTS[whatsapp]="WhatsApp Web en Brave"
COMMENTS[spotify]="Spotify en Brave"
COMMENTS[instagram]="Instagram en Brave"
COMMENTS[telegram]="Telegram Web en Brave"
COMMENTS[discord]="Discord en Brave"
COMMENTS[github]="GitHub en Brave"

# ------------------------------------------------------------
# DESCARGAR ICONOS
# ------------------------------------------------------------

echo
echo -e "${BOLD}----------------------------------------------${RESET}"
echo -e "${BOLD} DESCARGANDO ICONOS SVG${RESET}"
echo -e "${BOLD}----------------------------------------------${RESET}"
echo

for APP in youtube gmail whatsapp spotify instagram telegram discord github
do

    ICON_FILE="$ICON_DIR/$APP.svg"
    ICON_URL="$BASE_URL/$APP"

    echo -n "  $APP ... "

    if curl -fsSL "$ICON_URL" -o "$ICON_FILE"; then

        echo -e "${GREEN}OK${RESET}"

    else

        echo -e "${TERRACOTTA}ERROR${RESET}"

        rm -f "$ICON_FILE"

    fi

done

# ------------------------------------------------------------
# CREAR .DESKTOP
# ------------------------------------------------------------

echo
echo -e "${BOLD}----------------------------------------------${RESET}"
echo -e "${BOLD} CREANDO ARCHIVOS .DESKTOP${RESET}"
echo -e "${BOLD}----------------------------------------------${RESET}"
echo

for APP in youtube gmail whatsapp spotify instagram telegram discord github
do

    DESKTOP_FILE="$DESKTOP_DIR/$APP.desktop"
    ICON_FILE="$ICON_DIR/$APP.svg"

    cat > "$DESKTOP_FILE" <<EOF
[Desktop Entry]
Version=1.0
Type=Application

Name=${NAMES[$APP]}
GenericName=${NAMES[$APP]}
Comment=${COMMENTS[$APP]}

Exec=$BRAVE --app=${URLS[$APP]}
Icon=$ICON_FILE

Terminal=false
StartupNotify=true

Categories=Network;
EOF

    chmod +x "$DESKTOP_FILE"

    echo -e "  ${GREEN}[OK]${RESET} ${NAMES[$APP]}"

done

# ------------------------------------------------------------
# INSTALAR .DESKTOP PARA ROFI
# ------------------------------------------------------------

echo
echo -e "${BOLD}----------------------------------------------${RESET}"
echo -e "${BOLD} INSTALANDO APLICACIONES PARA ROFI${RESET}"
echo -e "${BOLD}----------------------------------------------${RESET}"
echo

for APP in youtube gmail whatsapp spotify instagram telegram discord github
do

    cp "$DESKTOP_DIR/$APP.desktop" \
       "$SYSTEM_APPS/$APP.desktop"

    chmod +x "$SYSTEM_APPS/$APP.desktop"

    echo -e "  ${GREEN}[OK]${RESET} ${NAMES[$APP]}"

done

# ------------------------------------------------------------
# ACTUALIZAR BASE DE DATOS
# ------------------------------------------------------------

if command -v update-desktop-database >/dev/null 2>&1; then

    info "Actualizando base de datos de aplicaciones..."

    update-desktop-database "$SYSTEM_APPS" 2>/dev/null || true

    ok "Base de datos actualizada"

else

    warn "update-desktop-database no está instalado."

fi

# ------------------------------------------------------------
# FINAL
# ------------------------------------------------------------

echo
echo -e "${BOLD}==============================================${RESET}"
echo -e "${GREEN} INSTALACIÓN COMPLETADA${RESET}"
echo -e "${BOLD}==============================================${RESET}"
echo

echo "Aplicaciones instaladas:"
echo

echo "  ▶ YouTube"
echo "  ✉ Gmail"
echo "  ◉ WhatsApp Web"
echo "  ♪ Spotify"
echo "  ◎ Instagram"
echo "  ✈ Telegram Web"
echo "  ◈ Discord"
echo "  ◆ GitHub"

echo
echo "Iconos:"
echo "  $ICON_DIR"

echo
echo "WebApps:"
echo "  $DESKTOP_DIR"

echo
echo "Rofi:"
echo "  $SYSTEM_APPS"

echo
echo -e "${BOLD}Prueba ahora:${RESET}"
echo
echo "  rofi -show drun"
echo

echo -e "${GREEN}Listo.${RESET}"
echo

# --- Bloque Git corregido (con colores seguros y contadores) ---


parse_git_block() {
    branch=$(git symbolic-ref --short HEAD 2>/dev/null)
    if [ -n "$branch" ]; then
        status=$(git status --porcelain 2>/dev/null)
        ahead=$(git rev-list --count @{u}..HEAD 2>/dev/null || echo 0)
        behind=$(git rev-list --count HEAD..@{u} 2>/dev/null || echo 0)

        # Normalizamos si están actualizados
        [ -z "$ahead" ] && ahead=0
        [ -z "$behind" ] && behind=0

        emoji=""
        color="$verde"   # verde limpio por defecto
        count=""

        if [ -n "$status" ]; then
            new=$(echo "$status" | grep "^??" | wc -l)
            modified=$(echo "$status" | grep "^[ M]" | wc -l)
            conflicts=$(echo "$status" | grep "^UU" | wc -l)

            total=$((new + modified + conflicts))
            count="$total"

            if [ "$conflicts" -gt 0 ]; then
                color="$rojo"   # rojo → conflictos
                emoji="❗"
            elif [ "$new" -gt 0 ]; then
                color="$rojo"   # rojo → archivos sin trackear
                emoji="❗"
            else
                color="$amarillo"   # amarillo → modificados
                emoji=""
            fi
        fi

        # Estado remoto
        [ "$ahead" -gt 0 ] && emoji="$emoji  󰦘  $ahead"
        [ "$behind" -gt 0 ] && emoji="$emoji  󰦗  $behind"

        # Imprime bloque con colores desde tus variables
        if [ -n "$count" ]; then
            echo "   ${branch}   $emoji "
        else
            echo "   ${branch}   $emoji"
        fi
    fi
}

Algo importante que veo en tu configuración
Ya tienes una estructura bastante grande de aliases y configuraciones personalizadas. No te recomendaría seguir metiendo todos los aliases directamente en .bashrc.

Ya tienes esta parte:

if [ -f ~/.bash_aliases ]; then
    . ~/.bash_aliases
fi
Eso significa que Bash ya está preparado para cargar ~/.bash_aliases.

Por tanto, una organización más limpia para tu RT-Linux-Theme sería:

~/
├── .bashrc
├── .bash_aliases
│
└── RT-Linux-Theme/
    └── webapps/
        ├── icons/
        └── applications/
Y poner todos los alias de WebApps en ~/.bash_aliases, dejando .bashrc para configuración del shell, prompt, Git, colores, etc.

Incluso podemos organizar tu .bash_aliases por bloques:

.bash_aliases
│
├── Archivos
├── Navegación
├── Multimedia
├── Sistema
├── Juegos
├── Git
├── ChatGPT
├── WebApps
└── RT-Linux-Theme
Eso sería bastante más limpio que seguir haciendo crecer el .bashrc que ya tienes.
